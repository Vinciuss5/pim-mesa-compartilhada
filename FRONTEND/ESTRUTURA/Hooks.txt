// src/hooks/useDoacao.js
import { useState, useCallback } from 'react';

const API_URL = import.meta.env.VITE_API_URL || 'http://localhost:5235/api';

export const useDoacao = () => {
    const [doacoesHistorico, setDoacoesHistorico] = useState([]);
    const [isLoading, setIsLoading] = useState(false);
    const [error, setError] = useState(null);

    const getHeaders = () => {
        const headers = { 'Content-Type': 'application/json' };
        const token = localStorage.getItem('token');
        if (token) headers['Authorization'] = `Bearer ${token}`;
        return headers;
    };

    // GET: Buscar histórico
    const fetchDoacoes = useCallback(async () => {
        setIsLoading(true);
        setError(null);
        try {
            const response = await fetch(`${API_URL}/doacao`, { headers: getHeaders() });
            if (!response.ok) throw new Error('Erro ao buscar o histórico de doações.');
            
            const data = await response.json();
            
            // "Achatando" a resposta: Cada item vira uma linha na tabela de histórico
            const historicoFormatado = [];
            data.forEach(doacao => {
                const dataFormatada = new Date(doacao.data).toLocaleDateString('pt-BR');
                doacao.itens.forEach(item => {
                    historicoFormatado.push({
                        idId: item.id, // ID interno do item (para chave do React)
                        idDisplay: `ENT-${doacao.id.toString().padStart(4, '0')}`, // Ex: ENT-0042
                        data: dataFormatada,
                        doador: doacao.nomeDoador,
                        produto: item.nomeProduto,
                        quantidade: `${item.quantidade} un`,
                        validade: item.dataValidade ? new Date(item.dataValidade).toLocaleDateString('pt-BR') : 'N/A'
                    });
                });
            });

            setDoacoesHistorico(historicoFormatado);
        } catch (err) {
            setError(err.message);
        } finally {
            setIsLoading(false);
        }
    }, []);

    // POST: Criar nova doação
    const createDoacao = async (payload) => {
        setIsLoading(true);
        setError(null);
        try {
            const response = await fetch(`${API_URL}/doacao`, {
                method: 'POST',
                headers: getHeaders(),
                body: JSON.stringify(payload)
            });

            if (!response.ok) {
                const errorData = await response.json().catch(() => ({}));
                throw new Error(errorData.message || 'Erro ao registrar entrada.');
            }

            const data = await response.json();
            return { success: true, data };
        } catch (err) {
            setError(err.message);
            return { success: false, message: err.message };
        } finally {
            setIsLoading(false);
        }
    };

    return { doacoesHistorico, isLoading, error, fetchDoacoes, createDoacao };
};

// src/hooks/useDoadores.js
import { useState, useCallback, useRef } from 'react';

// Cache em módulo: sobrevive entre montagens da página (mas some se recarregar o navegador)
let cacheDoadores = [];

export const useDoadores = () => {
  const [doadores, setDoadores] = useState(cacheDoadores);
  const [loading, setLoading] = useState(false);
  const [error, setError] = useState(null);
  const primeiraExecucao = useRef(true);

  const API_URL = import.meta.env.VITE_API_URL || 'http://localhost:5235/api';
  const DOADOR_URL = `${API_URL}/doador`;

  const fetchDoadores = useCallback(async (search = '') => {
    // Só mostra o loading se não tivermos nada em cache ainda
    if (cacheDoadores.length === 0) setLoading(true);
    setError(null);
    try {
      const url = search ? `${DOADOR_URL}?filtro=${encodeURIComponent(search)}` : DOADOR_URL;
      const response = await fetch(url);
      if (!response.ok) throw new Error('Erro ao carregar doadores');
      const data = await response.json();
      cacheDoadores = data;
      setDoadores(data);
    } catch (err) {
      setError(err.message);
    } finally {
      setLoading(false);
    }
  }, []);

  const createDoador = async (DoadorDTO) => {
    const response = await fetch(DOADOR_URL, {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify(DoadorDTO)
    });
    if (!response.ok) {
      const errorData = await response.json();
      throw new Error(errorData.message || 'Erro ao cadastrar');
    }
    return await response.json();
  };

  return { doadores, loading, error, fetchDoadores, createDoador, primeiraExecucao };
};

import { useState, useCallback } from 'react';

// Centralize variáveis de ambiente. Interfaces reais não usam localhost hardcoded em produção.
const API_BASE_URL = import.meta.env.VITE_API_URL 
  ? `${import.meta.env.VITE_API_URL}/usuario` 
  : 'http://localhost:5235/api/usuario'; 

export function useFuncionarios() {
  const [funcionarios, setFuncionarios] = useState([]);
  const [isLoading, setIsLoading] = useState(false);
  const [error, setError] = useState(null);

  const getHeaders = () => {
    return { 'Content-Type': 'application/json' };
  };

  const handleResponse = async (response) => {
    if (!response.ok) {
      const errorData = await response.json().catch(() => ({}));
      
      const serverMessage = errorData.message || 
                            (errorData.errors ? Object.values(errorData.errors).flat().join(", ") : null);
      
      throw new Error(serverMessage || 'Erro de comunicação com o servidor.');
    }
    return response.status === 204 ? null : response.json();
  };

  // 1. Listar (GET)
  const fetchFuncionarios = useCallback(async () => {
    setIsLoading(true);
    setError(null);
    try {
      const response = await fetch(API_BASE_URL, { headers: getHeaders() });
      const data = await handleResponse(response);
      setFuncionarios(data);
    } catch (err) {
      setError(err.message);
    } finally {
      setIsLoading(false);
    }
  }, []);

  // 2. Criar (POST)
  const createFuncionario = async (formData) => {
    setIsLoading(true);
    setError(null);

    const payload = {
      Nome: formData.nome.trim(),
      Email: formData.email.toLowerCase().trim(),
      CPF: formData.cpf.replace(/\D/g, ''),
      Telefone: formData.telefone?.replace(/\D/g, ''),
      Senha: formData.senha,
      Tipo: String(formData.tipo).toUpperCase(), 
      DataAdmissao: formData.dataAdmissao
    };

    try {
      const response = await fetch(API_BASE_URL, {
        method: 'POST',
        headers: getHeaders(),
        body: JSON.stringify(payload), 
      });

      // O backend retorna o UsuarioResponseDTO no método POST[cite: 2, 3]
      const novoUsuario = await handleResponse(response);
      
      // ERRO DE UX CORRIGIDO: Atualiza a lista local imediatamente sem novo GET
      setFuncionarios((prev) => [...prev, novoUsuario]);

      // CONTRATO CORRIGIDO: Retorna o mesmo padrão dos outros métodos
      return { success: true, data: novoUsuario };
    } catch (err) {
      setError(err.message);
      // Evite `throw err` se o componente espera checar `result.success`
      return { success: false, message: err.message };
    } finally {
      setIsLoading(false);
    }
  };

  // 3. Atualizar (PUT)
  const updateFuncionario = async (id, UsuarioDTO) => {
    setIsLoading(true);
    setError(null);
    
    try {
      const payload = {
        Nome: UsuarioDTO.nome,
        Email: UsuarioDTO.email,
        Telefone: UsuarioDTO.telefone,
        Tipo: UsuarioDTO.tipo,
        Ativo: UsuarioDTO.ativo
      };

      const response = await fetch(`${API_BASE_URL}/${id}`, {
        method: 'PUT',
        headers: getHeaders(),
        body: JSON.stringify(payload),
      });
      
      await handleResponse(response);
      
      setFuncionarios((prev) => 
        prev.map((f) => (f.id === id ? { ...f, ...payload } : f))
      );
      
      return { success: true };
    } catch (err) {
      setError(err.message);
      return { success: false, message: err.message };
    } finally {
      setIsLoading(false);
    }
  };

  // 4. Alternar Status (DELETE / PUT)
  const toggleStatus = async (funcionario) => {
    setIsLoading(true);
    setError(null);
    const novoStatus = !funcionario.ativo;

    try {
      let response;
      
      if (!novoStatus) {
        response = await fetch(`${API_BASE_URL}/${funcionario.id}`, {
          method: 'DELETE',
          headers: getHeaders(),
        });
      } else {
        const payload = {
            Nome: funcionario.nome,
            Email: funcionario.email,
            Telefone: funcionario.telefone,
            Tipo: funcionario.tipo,
            Ativo: true
        };
        response = await fetch(`${API_BASE_URL}/${funcionario.id}`, {
          method: 'PUT',
          headers: getHeaders(),
          body: JSON.stringify(payload),
        });
      }

      await handleResponse(response);

      setFuncionarios((prev) =>
        prev.map((f) => (f.id === funcionario.id ? { ...f, ativo: novoStatus } : f))
      );
      
      return { success: true };
    } catch (err) {
      setError(err.message);
      return { success: false, message: err.message };
    } finally {
      setIsLoading(false);
    }
  };

  return {
    funcionarios,
    isLoading,
    error,
    fetchFuncionarios,
    createFuncionario,
    updateFuncionario,
    toggleStatus,
  };
}


useRelatorios.js

  import { useState, useEffect, useMemo, useCallback } from 'react';
  import { useEstoque } from '../hooks/useEstoque';
  import { useDoacao } from '../hooks/useDoacao';
  import { useSolicitacao } from '../hooks/useSolicitacao';

  // Função auxiliar para converter 'DD/MM/YYYY' (gerado pelos seus hooks) em objeto Date
  const parseDataBR = (dataStr) => {
    if (!dataStr || dataStr === 'N/A') return null;
    if (dataStr.includes('/')) {
      const [dia, mes, ano] = dataStr.split('/');
      return new Date(ano, mes - 1, dia);
    }
    return new Date(dataStr); 
  };

  export const useRelatorios = () => {
    const [periodo, setPeriodo] = useState('MES_ATUAL'); // Valores possíveis: 'MES_ATUAL', 'ULTIMOS_30', 'ANO'
    const [isCarregandoInicial, setIsCarregandoInicial] = useState(true);

    // Importando seus hooks existentes
    const { estoque, fetchEstoque } = useEstoque();
    const { doacoesHistorico, fetchDoacoes } = useDoacao();
    const { solicitacoes, fetchSolicitacoes } = useSolicitacao();

    // Dispara as requisições em paralelo para carregar o Dashboard rápido
    const carregarDadosDashboard = useCallback(async () => {
      setIsCarregandoInicial(true);
      await Promise.all([
        fetchEstoque(),
        fetchDoacoes(),
        fetchSolicitacoes()
      ]);
      setIsCarregandoInicial(false);
    }, [fetchEstoque, fetchDoacoes, fetchSolicitacoes]);

    // Carrega ao montar o componente
    useEffect(() => {
      carregarDadosDashboard();
    }, [carregarDadosDashboard]);

    // ==========================================
    // LÓGICA DE NEGÓCIO E KPIs (Otimizado com useMemo)
    // ==========================================

    // 1. Alerta de Validade (Regra dos 3 Meses)
    const alertasValidade = useMemo(() => {
    const hoje = new Date();
    const dataLimite = new Date();
    dataLimite.setMonth(hoje.getMonth() + 3);

    const itensEmRisco = estoque.filter(item => {
      const dataValidade = parseDataBR(item.validade);
      
      // Converte a quantidade para número para garantir a comparação
      const quantidadeDisponivel = parseFloat(item.qtd) || 0;

      if (!dataValidade) return false;

      // NOVA REGRA: Só entra no relatório se a data estiver no prazo E a quantidade for maior que zero
      return (
        dataValidade >= hoje && 
        dataValidade <= dataLimite && 
        quantidadeDisponivel > 0 // <--- A mágica acontece aqui
      );
    });

    return itensEmRisco
      .sort((a, b) => parseDataBR(a.validade) - parseDataBR(b.validade))
      .map(item => ({
        id: item.id,
        produto: item.nome,
        lote: item.origem || 'N/D',
        validade: item.validade,
        quantidade: item.qtd
      }));
  }, [estoque]);

    // 2. Solicitações Pendentes e Histórico Recente
  const { ultimasSolicitacoes, solicitacoesPendentes } = useMemo(() => {
      const pendentesCount = solicitacoes.filter(s => 
        (s.status || s.Status) === 'PENDENTE'
      ).length;
      
      // ORDENAÇÃO: Garanta que o mais novo venha primeiro por data ou ID
      const ordenadas = [...solicitacoes].sort((a, b) => {
        return new Date(b.data || b.Data) - new Date(a.data || a.Data);
      });

      // UX DECISION: Se for Relatório, não use .slice(0, 5). 
      // Se for Dashboard, use, mas saiba que ele limitará a visão.
      // Vou remover o slice para você ver todos os dados.
      const processadas = ordenadas.map(sol => {
          // Tenta pegar a data de múltiplas fontes possíveis do JSON
          const dataBruta = sol.data || sol.Data || sol.dataSolicitacao;
          
          return {
              id: sol.id || sol.Id,
              solicitante: sol.nomeSolicitante || sol.NomeSolicitante || 'N/A',
              // Correção de parsing de data
              data: dataBruta ? new Date(dataBruta).toLocaleDateString('pt-BR') : 'Data Inválida',
              status: sol.status || sol.Status
          };
      });

      return { ultimasSolicitacoes: processadas, solicitacoesPendentes: pendentesCount };
  }, [solicitacoes]);

    // 3. Calculadora de Estoque Total (Peso x Qtd)
    const estoqueTotalKg = useMemo(() => {
      const total = estoque.reduce((acumulador, item) => {
        const peso = parseFloat(item.peso) || 0;
        const qtd = parseFloat(item.qtd) || 0;
        // Se tem peso cadastrado, multiplica. Se não, soma apenas a unidade/qtd
        const valorSoma = peso > 0 ? (peso * qtd) : qtd;
        return acumulador + valorSoma;
      }, 0);
      
      // Formata no padrão numérico brasileiro: 2.450,50
      return total.toLocaleString('pt-BR', { minimumFractionDigits: 2, maximumFractionDigits: 2 });
    }, [estoque]);

    // 4. Doações no Período Selecionado
    const doacoesFiltradasCount = useMemo(() => {
      const hoje = new Date();
      
      return doacoesHistorico.filter(doacao => {
        const dataDoacao = parseDataBR(doacao.data);
        if (!dataDoacao) return false;

        if (periodo === 'MES_ATUAL') {
          return dataDoacao.getMonth() === hoje.getMonth() && dataDoacao.getFullYear() === hoje.getFullYear();
        }
        if (periodo === 'ULTIMOS_30') {
          const trintaDiasAtras = new Date();
          trintaDiasAtras.setDate(hoje.getDate() - 30);
          return dataDoacao >= trintaDiasAtras && dataDoacao <= hoje;
        }
        if (periodo === 'ANO') {
          return dataDoacao.getFullYear() === hoje.getFullYear();
        }
        return true;
      }).length; 
    }, [doacoesHistorico, periodo]);

    // Objeto centralizado para o Dashboard
    const kpis = {
      solicitacoesPendentes,
      produtosProximosVencimento: alertasValidade.length,
      doacoesMes: doacoesFiltradasCount,
      estoqueTotalKg
    };

    return {
      kpis,
      alertasValidade,
      ultimasSolicitacoes,
      isLoading: isCarregandoInicial,
      periodo,
      setPeriodo,
      recarregarDados: carregarDadosDashboard
    };
  };

import { useState, useCallback } from 'react';

const API_URL = import.meta.env.VITE_API_URL || 'http://localhost:5235/api';

export const useSolicitacao = () => {
  const [solicitacoes, setSolicitacoes] = useState([]);
  const [loading, setLoading] = useState(false);
  const [error, setError] = useState(null);

  // GET: Lista todas as solicitações (Histórico)
  const fetchSolicitacoes = useCallback(async () => {
    setLoading(true);
    setError(null);
    try {
      const response = await fetch(`${API_URL}/solicitacao`);
      if (!response.ok) throw new Error('Erro ao carregar as solicitações.');
      
      const data = await response.json();
      setSolicitacoes(data);
    } catch (err) {
      console.error(err);
      setError(err.message);
    } finally {
      setLoading(false);
    }
  }, []);

  // POST: Cria uma nova solicitação
  const criarSolicitacao = useCallback(async (novaSolicitacao) => {
    setLoading(true);
    setError(null);
    try {
      const response = await fetch(`${API_URL}/solicitacao`, {
        method: 'POST',
        headers: {
          'Content-Type': 'application/json',
        },
        body: JSON.stringify(novaSolicitacao),
      });

      const data = await response.json();

      if (!response.ok) {
        // Captura a mensagem de erro de regra de negócio do C# (ex: saldo insuficiente)
        throw new Error(data.message || 'Erro ao criar solicitação.');
      }

      // Atualiza a lista local adicionando a nova solicitação no topo
      setSolicitacoes((prev) => [data, ...prev]);
      return { sucesso: true, data };
    } catch (err) {
      console.error(err);
      setError(err.message);
      return { sucesso: false, erro: err.message };
    } finally {
      setLoading(false);
    }
  }, []);

  // PATCH: Atualiza o status (PENDENTE, ENTREGUE, CANCELADO)
  const atualizarStatus = useCallback(async (id, novoStatus) => {
    setLoading(true);
    setError(null);
    try {
      const response = await fetch(`${API_URL}/solicitacao/${id}/status`, {
        method: 'PATCH',
        headers: {
          'Content-Type': 'application/json',
          // 'Authorization': `Bearer ${token}` // Descomente quando ativar o [Authorize]
        },
        body: JSON.stringify({ novoStatus }),
      });

      if (!response.ok) {
        const data = await response.json();
        throw new Error(data.message || 'Erro ao atualizar status.');
      }

      // Atualiza o status localmente para evitar um novo GET
      setSolicitacoes((prev) => 
        prev.map((sol) => sol.id === id ? { ...sol, status: novoStatus } : sol)
      );
      
      return { sucesso: true };
    } catch (err) {
      console.error(err);
      setError(err.message);
      return { sucesso: false, erro: err.message };
    } finally {
      setLoading(false);
    }
  }, []);

  return { 
    solicitacoes, 
    loading, 
    error, 
    fetchSolicitacoes, 
    criarSolicitacao, 
    atualizarStatus 
  };
};

// src/hooks/useSolicitantes.js
import { useState, useCallback, useRef } from 'react';

let cacheSolicitantes = [];

export const useSolicitantes = () => {
  const [solicitantes, setSolicitantes] = useState(cacheSolicitantes);
  const [loading, setLoading] = useState(false);
  const [error, setError] = useState(null);
  const primeiraExecucao = useRef(true);

  const API_URL = import.meta.env.VITE_API_URL || 'http://localhost:5235/api';
  const SOLICITANTE_URL = `${API_URL}/solicitante`;

  const fetchSolicitantes = useCallback(async (search = '') => {
    if (cacheSolicitantes.length === 0) setLoading(true);
    setError(null);
    try {
      const url = search 
        ? `${SOLICITANTE_URL}?search=${encodeURIComponent(search)}` 
        : SOLICITANTE_URL;

      const response = await fetch(url);
      if (!response.ok) throw new Error('Erro ao buscar solicitantes.');

      const data = await response.json();
      cacheSolicitantes = data;
      setSolicitantes(data);
    } catch (err) {
      setError(err.message);
      console.error("Erro na busca:", err);
    } finally {
      setLoading(false);
    }
  }, []);

  const createSolicitante = async (SolicitanteDTO) => {
    setLoading(true);
    setError(null);
    try {
      const response = await fetch(SOLICITANTE_URL, {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify(SolicitanteDTO),
      });

      if (!response.ok) {
        const errorData = await response.json();
        throw new Error(errorData.message || 'Erro ao cadastrar solicitante.');
      }

      const novoSolicitante = await response.json();
      cacheSolicitantes = [novoSolicitante, ...cacheSolicitantes];
      setSolicitantes(cacheSolicitantes);

      return { success: true, data: novoSolicitante };
    } catch (err) {
      setError(err.message);
      return { success: false, error: err.message };
    } finally {
      setLoading(false);
    }
  };

  return {
    solicitantes,
    loading,
    error,
    fetchSolicitantes,
    createSolicitante,
    primeiraExecucao
  };
};
