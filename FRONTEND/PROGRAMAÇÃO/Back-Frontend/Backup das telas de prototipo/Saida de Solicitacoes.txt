import React, { useState } from 'react';

export default function SaidaSolicitacoes() {
  const [isModalOpen, setIsModalOpen] = useState(false);
  
  // Estado para controlar qual linha da tabela está expandida (para ver os produtos)
  const [linhaExpandida, setLinhaExpandida] = useState(null);

  // Estados do Modal de Cadastro
  const [itensCarrinho, setItensCarrinho] = useState([]);
  const [produtoAtual, setProdutoAtual] = useState('');
  const [qtdAtual, setQtdAtual] = useState('');
  
  // Estados do Contexto da Solicitação
  const [solicitante, setSolicitante] = useState('');
  const [endereco, setEndereco] = useState('');
  const [dataPrevisao, setDataPrevisao] = useState('');
  const [observacao, setObservacao] = useState('');

  // Mock do Histórico com múltiplos produtos por solicitação
  const [historico] = useState([
    { 
      id: 'SAI-0102', data: '15/05/2026', solicitante: 'Creche Esperança', 
      endereco: 'Rua das Flores, 123 - Centro, Salto de Pirapora', 
      previsao: '18/05/2026', status: 'Pendente', observacao: 'Entregar na porta lateral.',
      produtos: [
        { nome: 'Arroz Branco 5kg', quantidade: '20.00 un' },
        { nome: 'Feijão Carioca 1kg', quantidade: '40.00 un' }
      ]
    },
    { 
      id: 'SAI-0101', data: '10/05/2026', solicitante: 'Associação Moradores', 
      endereco: 'Av. Principal, 1000 - Vila Nova', 
      previsao: '12/05/2026', status: 'Entregue', observacao: '',
      produtos: [
        { nome: 'Óleo de Soja 900ml', quantidade: '15.00 un' },
        { nome: 'Macarrão Espaguete', quantidade: '30.00 un' },
        { nome: 'Leite Integral', quantidade: '60.00 un' }
      ]
    },
    { 
      id: 'SAI-0100', data: '05/05/2026', solicitante: 'Família Silva', 
      endereco: 'Rua Projetada A, 45 - Bairro Alto', 
      previsao: '07/05/2026', status: 'Cancelado', observacao: 'Família mudou de cidade.',
      produtos: [
        { nome: 'Cesta Básica Padrão', quantidade: '1.00 un' }
      ]
    },
  ]);

  const toggleLinha = (id) => {
    if (linhaExpandida === id) {
      setLinhaExpandida(null); // Fecha se clicar novamente
    } else {
      setLinhaExpandida(id); // Abre a nova linha
    }
  };

  const adicionarProduto = (e) => {
    e.preventDefault();
    if (!produtoAtual || !qtdAtual) return;

    const novoItem = {
      id: Date.now(),
      nome: produtoAtual,
      quantidade: parseFloat(qtdAtual).toFixed(2)
    };
    setItensCarrinho([...itensCarrinho, novoItem]);
    setProdutoAtual('');
    setQtdAtual('');
  };

  const removerProduto = (id) => {
    setItensCarrinho(itensCarrinho.filter(item => item.id !== id));
  };

  const renderStatusBadge = (status) => {
    const statusConfig = {
      'Pendente': 'bg-amber-50 text-amber-700 border-amber-200',
      'Entregue': 'bg-emerald-50 text-emerald-700 border-emerald-200',
      'Cancelado': 'bg-red-50 text-red-700 border-red-200'
    };
    const style = statusConfig[status] || 'bg-gray-50 text-gray-700 border-gray-200';
    return (
      <span className={`px-2.5 py-1 rounded-full text-xs font-semibold border ${style}`}>
        {status}
      </span>
    );
  };

  return (
    <div className="min-h-screen bg-[#F8F9FA] text-[#2D3748] p-8 font-sans">
      
      {/* HEADER */}
      <div className="flex justify-between items-center mb-6">
        <div>
          <h1 className="text-2xl font-bold text-[#2D3748]">Saída de Solicitações</h1>
          <p className="text-sm text-[#718096] mt-1">Gerencie despachos, destinos e controle de entregas pendentes.</p>
        </div>
        <button 
          onClick={() => setIsModalOpen(true)}
          className="bg-[#007F45] hover:bg-[#006838] text-white px-4 py-2 rounded-md font-medium transition-colors flex items-center gap-2 text-sm shadow-sm"
        >
          <svg width="16" height="16" fill="none" stroke="currentColor" viewBox="0 0 24 24"><path strokeLinecap="round" strokeLinejoin="round" strokeWidth="2" d="M12 4v16m8-8H4"></path></svg>
          Nova Solicitação
        </button>
      </div>

      {/* BARRA DE PESQUISA ADICIONADA AQUI */}
      <div className="bg-white p-4 rounded-lg shadow-sm mb-6 border border-gray-100">
        <div className="relative w-full max-w-md">
          <div className="absolute inset-y-0 left-0 pl-3 flex items-center pointer-events-none">
            <svg className="h-4 w-4 text-[#718096]" fill="none" stroke="currentColor" viewBox="0 0 24 24"><path strokeLinecap="round" strokeLinejoin="round" strokeWidth="2" d="M21 21l-6-6m2-5a7 7 0 11-14 0 7 7 0 0114 0z"></path></svg>
          </div>
          <input 
            type="text" 
            placeholder="Buscar por ID ou Solicitante..." 
            className="block w-full pl-10 pr-3 py-2 border border-gray-200 rounded-md text-sm focus:outline-none focus:ring-1 focus:ring-[#007F45] focus:border-[#007F45]"
          />
        </div>
      </div>

      {/* TELA PRINCIPAL - TABELA COM ACCORDION */}
      <div className="bg-white rounded-lg shadow-sm border border-gray-100 overflow-hidden">
        <table className="w-full text-sm">
          <thead className="bg-gray-50 border-b border-gray-100 text-[#718096]">
            <tr>
              <th className="px-4 py-4 font-medium text-left w-20">ID</th>
              <th className="px-4 py-4 font-medium text-left">Data</th>
              <th className="px-4 py-4 font-medium text-left">Solicitante</th>
              <th className="px-4 py-4 font-medium text-left">Produtos</th>
              <th className="px-4 py-4 font-medium text-left hidden md:table-cell">Endereço de Entrega</th>
              <th className="px-4 py-4 font-medium text-left">Previsão</th>
              <th className="px-4 py-4 font-medium text-left">Status</th>
            </tr>
          </thead>
          <tbody className="divide-y divide-gray-100 text-[#2D3748]">
            {historico.map((item) => (
              <React.Fragment key={item.id}>
                {/* LINHA PRINCIPAL */}
                <tr className={`transition-colors ${linhaExpandida === item.id ? 'bg-green-50/30' : 'hover:bg-gray-50'}`}>
                  <td className="px-4 py-4 font-mono text-xs text-[#718096] whitespace-nowrap">{item.id}</td>
                  <td className="px-4 py-4 text-left whitespace-nowrap">{item.data}</td>
                  <td className="px-4 py-4 font-medium text-left">{item.solicitante}</td>
                  <td className="px-4 py-4 text-left">
                    {/* BOTÃO EXPANDIR - Substitui o "botão + solto" por um indicador de quantidade clicável */}
                    <button 
                      onClick={() => toggleLinha(item.id)}
                      className="flex items-center gap-1.5 text-[#007F45] hover:bg-[#007F45]/10 px-2 py-1 rounded border border-[#007F45]/30 transition-colors text-xs font-semibold"
                    >
                      {linhaExpandida === item.id ? '-' : '+'} {item.produtos.length} iten(s)
                    </button>
                  </td>
                  <td className="px-4 py-4 text-left hidden md:table-cell max-w-[200px] truncate" title={item.endereco}>
                    {item.endereco}
                  </td>
                  <td className="px-4 py-4 text-left whitespace-nowrap text-[#718096] font-medium">{item.previsao}</td>
                  <td className="px-4 py-4 text-left whitespace-nowrap">{renderStatusBadge(item.status)}</td>
                </tr>

                {/* SUB-LINHA EXPANSÍVEL (Detalhes do Pedido) */}
                {linhaExpandida === item.id && (
                  <tr className="bg-gray-50/80 border-b border-gray-200">
                    <td colSpan="7" className="p-0">
                      <div className="p-4 pl-12 pr-4 flex flex-col md:flex-row gap-6 animate-fadeIn">
                        
                        {/* Lista de Produtos na Expansão */}
                        <div className="flex-1 bg-white p-3 rounded border border-gray-200 shadow-sm">
                          <h4 className="text-xs font-bold text-[#718096] uppercase tracking-wider mb-2">Lista de Produtos</h4>
                          <ul className="divide-y divide-gray-100">
                            {item.produtos.map((prod, idx) => (
                              <li key={idx} className="py-1.5 flex justify-between items-center text-sm">
                                <span>{prod.nome}</span>
                                <span className="font-mono font-semibold text-[#007F45]">{prod.quantidade}</span>
                              </li>
                            ))}
                          </ul>
                        </div>

                        {/* Dados Complementares */}
                        <div className="flex-1 space-y-3">
                          <div>
                            <h4 className="text-xs font-bold text-[#718096] uppercase tracking-wider">Endereço Completo</h4>
                            <p className="text-sm text-[#2D3748] mt-1">{item.endereco}</p>
                          </div>
                          {item.observacao && (
                            <div>
                              <h4 className="text-xs font-bold text-[#718096] uppercase tracking-wider">Observações</h4>
                              <p className="text-sm text-[#2D3748] mt-1 bg-yellow-50 p-2 rounded border border-yellow-100">{item.observacao}</p>
                            </div>
                          )}
                        </div>
                        
                        {/* Ações Rápidas (Opcional para o futuro) */}
                        <div className="flex flex-col gap-2 justify-center border-l border-gray-200 pl-4">
                           {item.status === 'Pendente' && (
                             <button className="text-sm bg-white border border-[#007F45] text-[#007F45] hover:bg-[#007F45] hover:text-white px-3 py-1.5 rounded transition-colors font-medium">
                               Marcar Entregue
                             </button>
                           )}
                        </div>
                      </div>
                    </td>
                  </tr>
                )}
              </React.Fragment>
            ))}
          </tbody>
        </table>
      </div>

      {/* MODAL DE CADASTRO DE SOLICITAÇÃO */}
      {isModalOpen && (
        <div className="fixed inset-0 bg-[#0F172A]/40 backdrop-blur-sm flex items-center justify-center z-50 p-4">
          <div className="bg-white rounded-xl shadow-2xl w-full max-w-[800px] max-h-[90vh] flex flex-col overflow-hidden">
            
            <div className="flex justify-between items-center p-5 border-b border-gray-100 shrink-0">
              <h2 className="text-xl font-bold text-[#1A202C]">Nova Solicitação de Saída</h2>
              <button onClick={() => setIsModalOpen(false)} className="text-[#A0AEC0] hover:text-[#4A5568] transition-colors p-1 rounded-md hover:bg-gray-100">
                <svg width="24" height="24" fill="none" stroke="currentColor" viewBox="0 0 24 24"><path strokeLinecap="round" strokeLinejoin="round" strokeWidth="2" d="M6 18L18 6M6 6l12 12"></path></svg>
              </button>
            </div>

            <div className="flex-1 overflow-y-auto p-6 space-y-8 bg-[#FAFAFA]">
              
              {/* SEÇÃO 1: DADOS DO DESTINO */}
              <section className="bg-white p-5 rounded-lg border border-gray-200 shadow-sm">
                <h3 className="text-xs font-bold text-[#718096] uppercase tracking-wider mb-4 flex items-center gap-2">
                  <span className="w-5 h-5 rounded-full bg-gray-100 flex items-center justify-center text-[10px]">1</span>
                  Dados do Destino
                </h3>
                <div className="grid grid-cols-1 md:grid-cols-2 gap-4">
                  <div className="md:col-span-2">
                    <label className="block text-sm font-medium mb-1 text-[#4A5568]">Nome do Solicitante <span className="text-red-500">*</span></label>
                    <input type="text" value={solicitante} onChange={(e) => setSolicitante(e.target.value)} className="w-full border border-gray-300 rounded-md py-2 px-3 text-sm focus:ring-1 focus:ring-[#007F45] focus:outline-none" placeholder="Nome da instituição ou pessoa..." />
                  </div>
                  <div className="md:col-span-2">
                    <label className="block text-sm font-medium mb-1 text-[#4A5568]">Endereço de Entrega <span className="text-red-500">*</span></label>
                    <input type="text" value={endereco} onChange={(e) => setEndereco(e.target.value)} className="w-full border border-gray-300 rounded-md py-2 px-3 text-sm focus:ring-1 focus:ring-[#007F45] focus:outline-none" placeholder="Rua, número, bairro, cidade..." />
                  </div>
                  <div>
                    <label className="block text-sm font-medium mb-1 text-[#4A5568]">Data Prevista p/ Entrega <span className="text-red-500">*</span></label>
                    <input type="date" value={dataPrevisao} onChange={(e) => setDataPrevisao(e.target.value)} className="w-full border border-gray-300 rounded-md py-2 px-3 text-sm focus:ring-1 focus:ring-[#007F45] focus:outline-none" />
                  </div>
                  <div>
                    <label className="block text-sm font-medium mb-1 text-[#4A5568]">Observação (Opcional)</label>
                    <input type="text" value={observacao} onChange={(e) => setObservacao(e.target.value)} className="w-full border border-gray-300 rounded-md py-2 px-3 text-sm focus:ring-1 focus:ring-[#007F45] focus:outline-none" placeholder="Ex: Entregar de manhã..." />
                  </div>
                </div>
              </section>

              {/* SEÇÃO 2: PRODUTOS SOLICITADOS */}
              <section>
                <h3 className="text-xs font-bold text-[#718096] uppercase tracking-wider mb-4 flex items-center gap-2">
                  <span className="w-5 h-5 rounded-full bg-gray-100 flex items-center justify-center text-[10px]">2</span>
                  Montar Solicitação (Produtos)
                </h3>
                <div className="bg-white p-4 rounded-lg border border-[#E2E8F0] shadow-sm">
                  <form onSubmit={adicionarProduto} className="grid grid-cols-12 gap-3 items-end">
                    <div className="col-span-12 md:col-span-7">
                      <label className="block text-sm font-medium mb-1 text-[#4A5568]">Buscar Produto em Estoque <span className="text-red-500">*</span></label>
                      <input type="text" value={produtoAtual} onChange={(e) => setProdutoAtual(e.target.value)} className="w-full border border-gray-300 rounded-md py-2 px-3 text-sm focus:ring-1 focus:ring-[#007F45] focus:outline-none" placeholder="Digite o produto..." />
                    </div>
                    <div className="col-span-8 md:col-span-3">
                      <label className="block text-sm font-medium mb-1 text-[#4A5568]">Quantidade <span className="text-red-500">*</span></label>
                      <input type="number" step="0.01" min="0.01" value={qtdAtual} onChange={(e) => setQtdAtual(e.target.value)} className="w-full border border-gray-300 rounded-md py-2 px-3 text-sm focus:ring-1 focus:ring-[#007F45] focus:outline-none" placeholder="0.00" />
                    </div>
                    <div className="col-span-4 md:col-span-2">
                      <button type="submit" disabled={!produtoAtual || !qtdAtual} className="w-full bg-[#EDF2F7] hover:bg-[#E2E8F0] text-[#2D3748] font-medium py-2 rounded-md transition-colors text-sm disabled:opacity-50 disabled:cursor-not-allowed border border-gray-200">
                        Adicionar
                      </button>
                    </div>
                  </form>
                </div>
              </section>

              {/* SEÇÃO 3: LISTA DE SAÍDA */}
              <section>
                <div className="flex justify-between items-center mb-3">
                  <h3 className="text-xs font-bold text-[#718096] uppercase tracking-wider flex items-center gap-2">
                    <span className="w-5 h-5 rounded-full bg-gray-100 flex items-center justify-center text-[10px]">3</span>
                    Resumo do Despacho
                  </h3>
                  <span className="text-xs font-medium text-[#718096] bg-gray-100 px-2 py-1 rounded-full">{itensCarrinho.length} iten(s)</span>
                </div>
                
                <div className="bg-white border border-gray-200 rounded-lg overflow-hidden shadow-sm min-h-[120px]">
                  {itensCarrinho.length === 0 ? (
                    <div className="flex flex-col items-center justify-center h-32 text-[#A0AEC0]">
                      <svg className="w-8 h-8 mb-2 opacity-50" fill="none" stroke="currentColor" viewBox="0 0 24 24"><path strokeLinecap="round" strokeLinejoin="round" strokeWidth="1.5" d="M20 7l-8-4-8 4m16 0l-8 4m8-4v10l-8 4m0-10L4 7m8 4v10M4 7v10l8 4"></path></svg>
                      <p className="text-sm">Nenhum produto adicionado para despacho.</p>
                    </div>
                  ) : (
                    <ul className="divide-y divide-gray-100">
                      {itensCarrinho.map((item) => (
                        <li key={item.id} className="p-3 hover:bg-gray-50 flex justify-between items-center group">
                          <p className="text-sm font-semibold text-[#2D3748]">{item.nome}</p>
                          <div className="flex items-center gap-4">
                            <span className="text-sm font-mono font-medium text-[#007F45]">- {item.quantidade} un</span>
                            <button onClick={() => removerProduto(item.id)} className="text-red-400 hover:text-red-600 p-1.5 rounded-md hover:bg-red-50 transition-colors opacity-0 group-hover:opacity-100 focus:opacity-100" title="Remover item">
                              <svg width="18" height="18" fill="none" stroke="currentColor" viewBox="0 0 24 24"><path strokeLinecap="round" strokeLinejoin="round" strokeWidth="2" d="M19 7l-.867 12.142A2 2 0 0116.138 21H7.862a2 2 0 01-1.995-1.858L5 7m5 4v6m4-6v6m1-10V4a1 1 0 00-1-1h-4a1 1 0 00-1 1v3M4 7h16"></path></svg>
                            </button>
                          </div>
                        </li>
                      ))}
                    </ul>
                  )}
                </div>
              </section>

            </div>

            <div className="p-5 border-t border-gray-100 bg-white flex justify-between items-center shrink-0">
              <button onClick={() => setIsModalOpen(false)} className="text-[#718096] text-sm font-medium hover:text-[#2D3748] transition-colors">
                Cancelar Saída
              </button>
              <button 
                disabled={itensCarrinho.length === 0 || !solicitante || !endereco || !dataPrevisao}
                className="bg-[#007F45] hover:bg-[#006838] text-white px-6 py-2.5 rounded-md font-medium text-sm transition-all shadow-sm disabled:opacity-50 disabled:cursor-not-allowed flex items-center gap-2"
              >
                Registrar Solicitação
                <svg width="16" height="16" fill="none" stroke="currentColor" viewBox="0 0 24 24"><path strokeLinecap="round" strokeLinejoin="round" strokeWidth="2" d="M5 13l4 4L19 7"></path></svg>
              </button>
            </div>
            
          </div>
        </div>
      )}
    </div>
  );
}
